   805→/* --------------------------------------------------- Convenience selectors */
   806→
   807→export function getUniversityBySlug(slug: string): University | undefined {
   808→  return universities.find((u) => u.slug === slug);
   809→}
   810→
   811→export function getNeighbourhoodBySlug(slug: string): Neighbourhood | undefined {
   812→  return neighbourhoods.find((n) => n.slug === slug);
   813→}
   814→
   815→export function getPropertyBySlug(slug: string): Property | undefined {
   816→  return properties.find((p) => p.slug === slug);
   817→}
   818→
   819→export function getRoomById(id: string): RentableUnit | undefined {
   820→  return rooms.find((r) => r.id === id || r.slug === id);
   821→}
   822→
   823→export function getRoomsForProperty(propertyId: string): RentableUnit[] {
   824→  return rooms.filter((r) => r.propertyId === propertyId);
   825→}
   826→
   827→export function getFloorsForProperty(propertyId: string): Floor[] {
   828→  return floors.filter((f) => f.propertyId === propertyId);
   829→}
   830→
   831→export function getBookingsForStudent(studentId: string): Booking[] {
   832→  return bookings.filter((b) => b.studentId === studentId);
   833→}
   834→
   835→export function getPaymentsForStudent(studentId: string): Payment[] {
   836→  return payments.filter((p) => p.studentId === studentId);
   837→}
   838→
   839→export function getNotificationsForUser(userId: string): AppNotification[] {
   840→  return notifications
   841→    .filter((n) => n.userId === userId)
   842→    .sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt));
   843→}
